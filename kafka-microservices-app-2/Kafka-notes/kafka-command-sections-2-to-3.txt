# Section 2 Run the single Kafka server
---------------------------------------

I) To start single Kafka server
---------------------------------------

# To Generate cluster UUID
bin/kafka-storage.sh random-uuid 

# To format Kafka storage (Pass the cluster uuid)
bin/kafka-storage.sh format \ 
  --config config/server.properties \ 
  --cluster-id i5obkIXcSdKLvoyyYnp6-Q \ 
  --standalone 
 
 # To start the kafka server
 bin/kafka-server-start.sh config/server.properties 
 

II) To Start the Multiple Kafka Server
---------------------------------------

a) Step 1: Create configuration file. Copy the config/server.properties 3 times and rename to server-1.properties, server2-properties & server3.properties.

Copy server.properties three times: 

config/server-1.properties 
config/server-2.properties 
config/server-3.properties 

b) Step 2: Modify all 3 server properties as following

server-1.properties
-----------------

node.id=1  
listeners=PLAINTEXT://:9092,CONTROLLER://:9093 
controller.quorum.bootstrap.servers=localhost:9093,localhost:9095,localhost:9097 
advertised.listeners=PLAINTEXT://localhost:9092,CONTROLLER://localhost:9093 
log.dirs=/tmp/server-1/kraft-combined-logs 

server-2.properties
-----------------
node.id=2 
listeners=PLAINTEXT://:9094,CONTROLLER://:9095 
controller.quorum.bootstrap.servers=localhost:9093,localhost:9095,localhost:9097 
advertised.listeners=PLAINTEXT://localhost:9094,CONTROLLER://localhost:9095 
log.dirs=/tmp/server-2/kraft-combined-logs 

server-3.properties
-------------------

node.id=3 
listeners=PLAINTEXT://:9096,CONTROLLER://:9097 
controller.quorum.bootstrap.servers=localhost:9093,localhost:9095,localhost:9097 
advertised.listeners=PLAINTEXT://localhost:9096,CONTROLLER://localhost:9097 
log.dirs=/tmp/server-3/kraft-combined-logs 

c) Run these command to  generate UUID

./bin/kafka-server-stop.sh\n
rm -rf /tmp/server-1 /tmp/server-2 /tmp/server-3\n
export CLUSTER_ID=$(./bin/kafka-storage.sh random-uuid)\n
echo $CLUSTER_ID\n
DIR1=$(./bin/kafka-storage.sh random-uuid)
DIR2=$(./bin/kafka-storage.sh random-uuid)\n
DIR3=$(./bin/kafka-storage.sh random-uuid)\n
DIR1=R0k8aW5cS3y4j3mR8F2Yyw\nDIR2=3Zr1K5w6QmC4Jb5F9xX2Gg\nDIR3=F8KxQ4Tn2M1wR9sC6yZp0Q\n
DIR1=$(./bin/kafka-storage.sh random-uuid)
DIR2=$(./bin/kafka-storage.sh random-uuid)\n
DIR3=$(./bin/kafka-storage.sh random-uuid)\n
echo $DIR1 $DIR2 $DIR3

 d) To  Format the storage directories for each broker with the cluster ID and initial controller quorum
./bin/kafka-storage.sh format \\n  -t "$CLUSTER_ID" \\n  -c config/server-1.properties \\n  --initial-controllers "1@localhost:9093:$DIR1,2@localhost:9095:$DIR2,3@localhost:9097:$DIR3"\n
./bin/kafka-storage.sh format \\n  -t "$CLUSTER_ID" \\n  -c config/server-2.properties \\n  --initial-controllers "1@localhost:9093:$DIR1,2@localhost:9095:$DIR2,3@localhost:9097:$DIR3"\n
./bin/kafka-storage.sh format \\n  -t "$CLUSTER_ID" \\n  -c config/server-3.properties \\n  --initial-controllers "1@localhost:9093:$DIR1,2@localhost:9095:$DIR2,3@localhost:9097:$DIR3"\n
 
 e)  Start the servers
./bin/kafka-server-start.sh config/server-1.properties
./bin/kafka-server-start.sh config/server-2.properties
./bin/kafka-server-start.sh config/server-3.properties

f) Create Topics
